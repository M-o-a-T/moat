#!/usr/bin/python3

from moat.link.client import Link
from moat.lib.path import P
from moat.lib.config import CFG
import anyio

VALVE=P("home.ass.dyn.switch.plug-n5.relay.cmd")
VALVES=P("home.ass.dyn.switch.plug-n5.relay.state")
TEMP=P("legacy.heat.s.pump.temp.in"),P("legacy.heat.s.pump.temp.out")
FLOW=P("legacy.heat.s.pump.pid")
STATE=P("legacy.heat.s.pump.state")

c_flow=1/16  # more than 5%
t1=5
t2=60
temp1=8
temp2=10

async def main():
    cfg = CFG().moat
    qw,qr=anyio.create_memory_object_stream(2)

    async def poll(name, path):
        async with li.d_watch(path) as mon:
            async for val in mon:
                await qw.send((name,val))

    temp=[None]*len(TEMP)
    mtemp=15
    flow=False
    state=False
    valve=False
    f_ext = False

    mine=False
    at_work=False

    async with Link(cfg.link) as li, anyio.create_task_group() as tg:
        for i,tt in enumerate(TEMP):
            tg.start_soon(poll,i,tt)
        tg.start_soon(poll,"f",FLOW)
        tg.start_soon(poll,"s",STATE)
        tg.start_soon(poll,"v",VALVES)

        async def do_work():
            nonlocal at_work,mtemp,mine,valve
            at_work=True
            try:
                print("OPEN F")
                await li.d_set(VALVE,True)
                while mtemp < temp2:
                    mine=True
                    await li.d_set(FLOW,c_flow)
                    await anyio.sleep(t1)
                    mine=False
                    if f_ext:
                        return
                    await li.d_set(FLOW,0)
                    for _ in range(10):
                        await anyio.sleep(t2/10)
                        if mtemp >= temp2 or not valve:
                            return


            finally:
                at_work=False
                mine=False
                if f_ext:
                    print("NoCLOSE F")
                else:
                    print("CLOSE F")
                    await li.d_set(VALVE,False)

        async def chk_ext():
            nonlocal state,flow,f_ext
            if state and flow>0 and not f_ext:
                f_ext=True
                print("OPEN")
                await li.d_set(VALVE,True)
            elif f_ext and not (state and flow):
                f_ext=False
                print("CLOSE")
                if not mine:
                    await li.d_set(VALVE,False)

        while True:
            async for n,x in qr:
                if isinstance(n,int):
                    temp[n]=x
                    if None in temp:
                        continue
                    x=min(temp)
                    if x<temp1 and mtemp>=temp1 and not at_work:
                        tg.start_soon(do_work)
                    mtemp=x

                elif n=="s":
                    state=x!=0
                    await chk_ext()

                elif n=="v":
                    valve=x

                elif n=="f":
                    if flow==(x>0):
                        continue
                    flow=x>0
                    if mine and flow and x != c_flow:
                        f_ext=True
                    else:
                        await chk_ext()

                print(f"{mtemp:.1f} {'F' if flow else ''}{'M' if mine else ''}{'X' if f_ext else ''}{'W' if at_work else ''}")


anyio.run(main, backend="trio")
